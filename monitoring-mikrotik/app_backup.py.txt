# =========================
# KIRIM KE GOOGLE SPREADSHEET
# =========================

def send_to_spreadsheet():

    data = {
        "router_name": router_data["name"],
        "ip": router_data["ip"],
        "version": router_data["version"],
        "cpu": router_data["cpu"],
        "memory": router_data["memory"],
        "uptime": router_data["uptime"],
        "download": traffic_data["download"],
        "upload": traffic_data["upload"]
    }

    try:
        response = requests.post(
            GOOGLE_SCRIPT_URL,
            json=data,
            timeout=10
        )

        print("====================================")
        print("STATUS CODE:", response.status_code)
        print("DATA DIKIRIM KE GOOGLE SPREADSHEET")
        print(response.text)
        print("====================================")

        return True

    except Exception as e:

        print("====================================")
        print("GAGAL MENGIRIM KE SPREADSHEET")
        print(str(e))
        print("====================================")

        return False

# LOGIN
# =========================

@app.route("/login", methods=["GET", "POST"])
def login():

    if request.method == "POST":

        username = request.form.get("username")
        password = request.form.get("password")

        if username == "admin" and password == "admin123":

            session["logged_in"] = True

            return redirect(url_for("dashboard"))

        return render_template(
            "login.html",
            error="Username atau password salah!"
        )

    return render_template("login.html")


# =========================
# LOGOUT
# =========================

@app.route("/logout", methods=["GET"])
def logout():

    session.clear()

    return redirect(url_for("login"))


# =========================
# DASHBOARD
# =========================

@app.route("/", methods=["GET"])
def dashboard():

    if not session.get("logged_in"):
        return redirect(url_for("login"))

    return render_template(
        "index.html",
        router=router_data,
        traffic=traffic_data
    )


# =========================
# UPDATE IP
# =========================

@app.route("/update-ip", methods=["POST"])
def update_ip():

    if not session.get("logged_in"):
        return redirect(url_for("login"))

    new_ip = request.form.get("ip", "").strip()

    if new_ip:
        router_data["ip"] = new_ip

    return redirect(url_for("dashboard"))


# =========================
# KIRIM KE SPREADSHEET

@app.route("/send-to-sheet", methods=["POST"])
def send_to_sheet():
