            link.getAttribute(

    });

/* =====================================================
   GRAFIK CPU
===================================================== */

function drawCPUChart() {
    const canvas = document.getElementById("cpuChart");

    if (!canvas) return;

    const ctx = canvas.getContext("2d");

    canvas.width = canvas.clientWidth;
    canvas.height = canvas.clientHeight;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    const values = cpuHistory;

    if (values.length < 2) return;

    const padding = 35;
    const graphWidth = canvas.width - padding * 2;
    const graphHeight = canvas.height - padding * 2;

    // Garis grid
    ctx.strokeStyle = "rgba(120, 130, 140, 0.15)";
    ctx.lineWidth = 1;

    for (let i = 0; i <= 5; i++) {
        const y = padding + (graphHeight / 5) * i;

        ctx.beginPath();
        ctx.moveTo(padding, y);
        ctx.lineTo(canvas.width - padding, y);
        ctx.stroke();
    }

    // Garis grafik
    ctx.beginPath();

    values.forEach((value, index) => {
        const x =
            padding +
            (graphWidth / (values.length - 1)) * index;

        const y =
            padding +
            graphHeight -
            (value / 100) * graphHeight;

        if (index === 0) {
            ctx.moveTo(x, y);
        } else {
            ctx.lineTo(x, y);
        }
    });

    ctx.strokeStyle = "#39ff88";
    ctx.lineWidth = 3;
    ctx.stroke();

    // Titik grafik
    values.forEach((value, index) => {
        const x =
            padding +
            (graphWidth / (values.length - 1)) * index;

        const y =
            padding +
            graphHeight -
            (value / 100) * graphHeight;

        ctx.beginPath();
        ctx.arc(x, y, 4, 0, Math.PI * 2);

        ctx.fillStyle = "#39ff88";
        ctx.fill();

        ctx.strokeStyle = "#07130d";
        ctx.lineWidth = 2;
        ctx.stroke();
    });
}



/* =====================================================
   GRAFIK TRAFFIC
===================================================== */

function drawTrafficChart() {
    const canvas = document.getElementById("trafficChart");

    if (!canvas) return;

    const ctx = canvas.getContext("2d");

    canvas.width = canvas.clientWidth;
    canvas.height = canvas.clientHeight;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    if (downloadHistory.length < 2) return;

    const padding = 35;
    const graphWidth = canvas.width - padding * 2;
    const graphHeight = canvas.height - padding * 2;

    let maxValue = Math.max(
        ...downloadHistory,
        ...uploadHistory
    );

    if (maxValue < 10) {
        maxValue = 10;
    }

    maxValue = Math.ceil(maxValue / 10) * 10;

    // Grid
    ctx.strokeStyle = "rgba(120, 130, 140, 0.15)";
    ctx.lineWidth = 1;

    for (let i = 0; i <= 5; i++) {
        const y = padding + (graphHeight / 5) * i;

        ctx.beginPath();
        ctx.moveTo(padding, y);
        ctx.lineTo(canvas.width - padding, y);
        ctx.stroke();
    }

    function drawLine(data, lineColor) {
        if (data.length < 2) return;

        ctx.beginPath();

        data.forEach((value, index) => {

            const x =
                padding +
                (graphWidth / (data.length - 1)) * index;

            const y =
                padding +
                graphHeight -
                (value / maxValue) * graphHeight;

            if (index === 0) {
                ctx.moveTo(x, y);
            } else {
                ctx.lineTo(x, y);
            }
        });

        ctx.strokeStyle = lineColor;
        ctx.lineWidth = 3;
        ctx.stroke();

        // titik
        data.forEach((value, index) => {

            const x =
                padding +
                (graphWidth / (data.length - 1)) * index;

            const y =
                padding +
                graphHeight -
                (value / maxValue) * graphHeight;

            ctx.beginPath();
            ctx.arc(x, y, 4, 0, Math.PI * 2);

            ctx.fillStyle = lineColor;
            ctx.fill();
        });
    }

    // Download
    drawLine(downloadHistory, "#39ff88");

    // Upload
    drawLine(uploadHistory, "#00d9ff");
}

    try {

        const response = await fetch("/api/status");

        if (!response.ok) {

        }
